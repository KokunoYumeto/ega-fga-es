\subsection{Morfismos lisos, morfismos no ramificados, morfismos étale}
\label{subsection:IV.17.3}

\begin{definition}[17.3.1]
\label{IV.17.3.1}
Se dice que un morfismo $f:X\to Y$ es liso \textup{(resp.}\ no ramificado, \textup{o} neto\footnote{Las palabras «neto» y «formalmente neto» parecen claramente preferibles a la terminología consagrada de «no ramificado» (resp. «formalmente no ramificado») y se emplearán casi exclusivamente a partir del cap. V. En este capítulo hemos conservado la terminología antigua para no entrar en conflicto con (\textbf 0, 19.10).}, \textup{resp.}\ étale\textup{)} si es localmente de presentación finita y formalmente liso \textup{(resp.}\ formalmente no ramificado, \textup{resp.}\ formalmente étale\textup{)}.
\end{definition}

Se dice también entonces que $X$ es \emph{liso} (resp. \emph{no ramificado} o \emph{neto}, resp. \emph{étale}) \emph{sobre} $Y$.

Veremos más adelante (17.5.2) que esta definición de morfismo liso coincide con la ya dada en (6.8.1); hasta entonces, utilizaremos exclusivamente la definición de (17.3.1).

Es claro que decir que $f$ es étale significa que es \emph{a la vez} liso y no ramificado.

\begin{remarks}[17.3.2]
\label{IV.17.3.2}
\begin{enumerate}
  \item[(i)] Obsérvese que la definición (17.3.1) puede expresarse únicamente mediante el funtor
  \[
    Y'\leadsto\Hom_Y(Y',X)
  \]
  considerado en (17.1.2, (iii)), pues decir que $f$ es localmente de presentación finita equivale a decir que el funtor anterior \emph{conmuta con los límites proyectivos de esquemas afines} (8.14.2).
  \item[(ii)] Sean $A$ un anillo, $B$ una $A$-álgebra. Se dice que $B$ es una $A$-álgebra \emph{lisa} (resp. \emph{no ramificada}, resp. \emph{étale}) si el morfismo correspondiente $\Spec(B)\to\Spec(A)$ es liso (resp. no ramificado, resp. étale). Equivale a decir que $B$ es una $A$-álgebra \emph{de presentación finita} (1.4.6) y formalmente lisa (resp. formalmente no ramificada, resp. formalmente étale) para las topologías discretas.
  \item[(iii)] De (17.1.6) y de la definición de morfismo localmente de presentación finita (1.4.2) resulta que las nociones de morfismo liso, no ramificado y étale son \emph{locales sobre $X$ y sobre $Y$}.
\end{enumerate}
\end{remarks}

\begin{proposition}[17.3.3]
\label{IV.17.3.3}
\begin{enumerate}
  \item[\textup{(i)}] Una inmersión abierta es étale. Para que una inmersión sea no ramificada, es necesario y suficiente que sea localmente de presentación finita.
  \item[\textup{(ii)}] La composición de dos morfismos lisos \textup{(resp.}\ no ramificados, \textup{resp.}\ étale\textup{)} es lisa \textup{(resp.}\ no ramificada, \textup{resp.}\ étale\textup{)}.
  \item[\textup{(iii)}] Si $f:X\to Y$ es un $S$-morfismo liso \textup{(resp.}\ no ramificado, \textup{resp.}\ étale\textup{)}, lo mismo ocurre con $f_{(S')}:X_{(S')}\to Y_{(S')}$ para toda extensión $S'\to S$ del preesquema base.
  \item[\textup{(iv)}] Si $f:X\to X'$ y $g:Y\to Y'$ son dos $S$-morfismos lisos \textup{(resp.}\ no ramificados, \textup{resp.}\ étale\textup{)}, lo mismo ocurre con $f\times_S g:X\times_S Y\to X'\times_S Y'$.
\end{enumerate}

\begin{enumerate}
  \item[\textup{(v)}] Sean $f:X\to Y$, $g:Y\to Z$ dos morfismos; si $g$ es localmente de tipo finito y si $g\circ f$ es no ramificado, entonces $f$ es no ramificado.
\end{enumerate}
\end{proposition}

Esto resulta inmediatamente de (1.4.3) y de (17.1.3).

\begin{proposition}[17.3.4]
\label{IV.17.3.4}
Sean $f:X\to Y$, $g:Y\to Z$ dos morfismos, y supongamos que $g$ es no ramificado. Entonces, si $g\circ f$ es liso \textup{(resp.}\ no ramificado, \textup{resp.}\ étale\textup{)}, lo mismo ocurre con $f$.
\end{proposition}

En efecto, como $g$ y $g\circ f$ son localmente de presentación finita, lo mismo ocurre con $f$ (1.4.4.3, (v)); la conclusión resulta, pues, de (17.1.4) y (17.1.3, (v)).

\begin{corollary}[17.3.5]
\label{IV.17.3.5}
Supongamos que $g$ es étale; entonces, para que $f$ sea liso \textup{(resp.}\ no ramificado, \textup{resp.}\ étale\textup{)}, es necesario y suficiente que $g\circ f$ lo sea.
\end{corollary}

Esto resulta de (17.3.4) y de (17.3.3, (ii)).

\begin{proposition}[17.3.6]
\label{IV.17.3.6}
Sean $g:Y\to S$, $h:X\to S$ dos morfismos localmente de presentación finita. Para que un $S$-morfismo $f:X\to Y$ sea no ramificado, es necesario y suficiente que el homomorfismo canónico \textup{(16.4.19)}
\[
  f^*(\varOmega_{Y/S}^1)\longrightarrow\varOmega_{X/S}^1
\]
sea sobreyectivo.
\end{proposition}

Como $f$ es entonces localmente de presentación finita (1.4.4.3, (v)), la proposición resulta inmediatamente de (17.2.2).

\begin{definition}[17.3.7]
\label{IV.17.3.7}
Sea $f:X\to Y$ un morfismo. Se dice que $f$ es liso \textup{(resp.}\ no ramificado, \textup{resp.}\ étale\textup{)} en un punto $x\in X$ si existe una vecindad abierta $U$ de $x$ en $X$ tal que la restricción $f|U$ sea un morfismo liso \textup{(resp.}\ no ramificado, \textup{resp.}\ étale\textup{)} de $U$ en $Y$.
\end{definition}

Se dice también entonces que $X$ es \emph{liso} (resp. \emph{no ramificado}, resp. \emph{étale}) \emph{sobre} $Y$ \emph{en el punto} $x$.

Teniendo en cuenta la observación (17.3.2, (iii)), equivale a decir que $f$ es un morfismo liso (resp. no ramificado, resp. étale) o que es liso (resp. no ramificado, resp. étale) en cada punto de $X$.

Es claro que el conjunto de los puntos de $X$ en los que un morfismo $f:X\to Y$ es liso (resp. no ramificado, resp. étale) es \emph{abierto} en $X$.

\begin{proposition}[17.3.8]
\label{IV.17.3.8}
Para todo preesquema $Y$ y todo $\sh{O}_Y$-módulo localmente libre $\sh{E}$ de tipo finito, el preesquema fibrado vectorial $\mathbf{V}(\sh{E})$ \textup{(\textbf{II}, 1.7.8)} es un $Y$-preesquema liso.
\end{proposition}

En efecto (17.3.2, (iii)), se puede restringir al caso en que $Y=\Spec(A)$ es afín y $\mathbf{V}(\sh{E})=\Spec(A[T_1,\ldots,T_r])$; como $A[T_1,\ldots,T_r]$ es una $A$-álgebra formalmente lisa para las topologías discretas (\textbf 0, 19.3.2) y de presentación finita, esto demuestra la proposición (17.3.2, (ii)).

\begin{corollary}[17.3.9]
\label{IV.17.3.9}
Bajo las hipótesis de \textup{(17.3.8)}, el preesquema fibrado proyectivo $\mathbf{P}(\sh{E})$ \textup{(\textbf{II}, 4.1.1)} es un $Y$-preesquema liso.
\end{corollary}

También se puede restringir al caso en que $Y=\Spec(A)$ es afín y $\mathbf{P}(\sh{E})=\mathbf{P}^r_Y$. Se sabe entonces (\textbf{II}, 2.3.14) que se tiene un recubrimiento abierto finito de $\mathbf{P}^r_A$ tomando los $D_+(T_i)$ ($0\leq i\leq r$), iguales respectivamente a los espectros de los anillos $S_{(f)}$, donde se sustituye $S$ por $A[T_0,T_1,\ldots,T_r]$ y $f$ por $T_i$; pero de la definición de $S_{(f)}$ (\textbf{II}, 2.2.1) resulta inmediatamente que este anillo, en el caso considerado, es isomorfo a $A[T_0,\ldots,T_{i-1},T_{i+1},\ldots,T_r]$; por tanto, el corolario resulta de (17.3.8).
