\begin{proposition}[6.5.13]
\label{III.6.5.13}
Sean $X$ e $Y$ dos $S$-preesquemas, y supongamos que se cumple
una de las condiciones siguientes:
\begin{enumerate}
  \item[\textup{(i)}] $X$ y $Z=X\times_SY$ son localmente noetherianos, y
    $X$ es plano sobre $S$;
  \item[\textup{(ii)}] $S$ y $X$ son localmente noetherianos, y $Y$ es de
    tipo finito sobre $S$.
\end{enumerate}
Sea $\sh P_\bullet$ (respectivamente $\sh Q_\bullet$) un complejo
acotado inferiormente de $\sh O_X$-módulos cuasicoherentes
(respectivamente de $\sh O_Y$-módulos cuasicoherentes).
Supongamos además que, para todo $n$,
$\sh H_n(\sh P_\bullet)$ (respectivamente
$\sh H_n(\sh Q_\bullet)$) es un $\sh O_X$-módulo
(respectivamente un $\sh O_Y$-módulo) de tipo finito. Entonces
\[
  \shTor_n^S(\sh P_\bullet,\sh Q_\bullet)
\]
es un $\sh O_Z$-módulo coherente para todo $n$.
\end{proposition}

\begin{proof}
Como $\sh P_\bullet$ y $\sh Q_\bullet$ están acotados
inferiormente, la sucesión espectral
$''\sh E(\sh P_\bullet,\sh Q_\bullet)$ es birregular por
\sref{III.6.5.4}. Tanto en el caso (i) como en el caso (ii), $Z$
es localmente noetheriano, de modo que \sref[0]{0.11.1.8} reduce
la afirmación a la coherencia de los términos
$''\sh E_{pq}^2$. La hipótesis de tipo finito sobre los módulos de
homología y la fórmula \sref{III.6.5.4.2} reducen la proposición
al caso particular en que $\sh P_\bullet$ y $\sh Q_\bullet$ están
concentrados en grado cero, es decir, al corolario siguiente.
\end{proof}

\begin{corollary}[6.5.14]
\label{III.6.5.14}
Supongamos que se cumple una de las condiciones \emph{(i)} y
\emph{(ii)} de \sref{III.6.5.13}, y sea $\sh F$
(respectivamente $\sh G$) un $\sh O_X$-módulo cuasicoherente
(respectivamente un $\sh O_Y$-módulo cuasicoherente) de tipo
finito. Entonces $\shTor_n^S(\sh F,\sh G)$ es un
$\sh O_Z$-módulo coherente para todo $n$.
\end{corollary}

\begin{proof}
La cuestión es local en $X$ e $Y$, por lo que podemos suponer que
$S$, $X$ e $Y$ son afines.

\textnormal{(i)} Bajo las hipótesis de (i), $S$, $X$ y $Z$ son
noetherianos. Existe, por tanto, una resolución localmente libre
$\sh L_\bullet$ de $\sh F$, formada por $\sh O_X$-módulos de
tipo finito (\sref[I]{I.1.3.7}). Como $X$ es plano sobre $S$,
\sref{III.6.3.4} da
\[
  \shTor_n^S(\sh F,\sh G)
  =
  \sh H_n(\sh L_\bullet\otimes_S\sh G).
\]
Los módulos $\sh L_i\otimes_S\sh G$ son
$\sh O_Z$-módulos cuasicoherentes de tipo finito
(\sref[I]{I.9.1.1}), luego son coherentes. Se sigue que
$\sh H_n(\sh L_\bullet\otimes_S\sh G)$ es coherente
(\sref[0]{0.5.3.4}).

\textnormal{(ii)} Supongamos ahora que se cumple la condición
(ii). Como $A(Y)$ es cociente de una $A(S)$-álgebra de polinomios
en un número finito de indeterminadas (\sref[I]{I.6.3.3}), $Y$ es
un subpreesquema cerrado de un $S$-esquema afín $Y'$ plano y de
tipo finito sobre $S$. El esquema $Y'$ es noetheriano
(\sref[I]{I.6.3.7}), por lo que existe una resolución localmente
libre $\sh M_\bullet$ de $j_*\sh G$ por
$\sh O_{Y'}$-módulos de tipo finito, donde $j:Y\to Y'$ es la
inmersión canónica. Por \sref{III.6.5.12},
$\shTor_n^S(\sh F,\sh G)$ es la imagen inversa por $1\times j$
del $\sh O_{Z'}$-módulo
\[
  \sh H_n(\sh F\otimes_S\sh M_\bullet),
  \qquad Z'=X\times_SY'.
\]
Como en (i), los $\sh F\otimes_S\sh M_i$ son
$\sh O_{Z'}$-módulos coherentes, y \sref[0]{0.5.3.4} da de nuevo
la conclusión.
\end{proof}

una sucesión exacta de $\sh{O}_{\widehat{X}}$-módulos coherentes.
Si $\sh{G}$ y $\sh{H}$ son algebraizables, entonces $\sh{F}$ es
algebraizable.
\end{lemma}

\begin{proof}
Supongamos que
\[
  \sh{G}=\widehat{\sh{B}},
  \qquad
  \sh{H}=\widehat{\sh{C}},
\]
donde $\sh{B}$ y $\sh{C}$ son $\sh{O}_X$-módulos coherentes cuyos
soportes son propios sobre $Y$. El módulo $\sh{F}$ define canónicamente
un elemento del $A$-módulo
\[
  \Ext^1_{\sh{O}_{\widehat{X}}}
  (\widehat{X};\widehat{\sh{B}},\widehat{\sh{C}})
\]
\sref[0]{0.12.3.2}. Las hipótesis implican que este $A$-módulo es
canónicamente isomorfo a
\[
  \Ext^1_{\sh{O}_X}(X;\sh{B},\sh{C})
\]
\sref{III.4.5.2}. Existe, por tanto, una sucesión exacta
\[
  0\longrightarrow\sh{C}\longrightarrow\sh{A}
  \longrightarrow\sh{B}\longrightarrow0
\]
de $\sh{O}_X$-módulos coherentes cuya clase canónica se envía en la
clase correspondiente a $\sh{F}$. Por definición, teniendo en cuenta
\sref[I]{I.10.8.8}, (ii), esto significa que $\sh{F}$ es isomorfo a
$\widehat{\sh{A}}$. El lema se deduce, pues
$\Supp(\sh{A})$ está contenido en
$\Supp(\sh{B})\cup\Supp(\sh{C})$ y es, por tanto, propio sobre $Y$.
\end{proof}

\begin{corollary}[5.3.2]

$u:\sh{F}\to\sh{G}$ un homomorfismo de
$\sh{O}_{\widehat{X}}$-módulos coherentes. Si $\sh{G}$, $\Ker(u)$ y
$\Coker(u)$ son algebraizables, entonces $\sh{F}$ es algebraizable.
\end{corollary}

\begin{proof}
El Lema \sref{III.5.2.2}, aplicado a
$\sh{G}\to\Coker(u)$, muestra que $\Im(u)$ es algebraizable. Basta entonces
aplicar el Lema \sref{III.5.3.1} a la sucesión exacta

  \longrightarrow\Im(u)\longrightarrow0.
\]
\end{proof}

\begin{theorem}[5.4.1]
\label{III.5.4.1}
Sea $A$ un anillo noetheriano ádico, sea $\mathfrak{J}$ un ideal
de definición de $A$, y pongamos
\[
  S=\Spec(A),\qquad S'=\mathrm{V}(\mathfrak{J}).
\]
Sea $u:X\to S$ un morfismo propio y sea $v:Y\to S$ un
morfismo separado de tipo finito. Sean
$\widehat{S}$, $\widehat{X}$ y $\widehat{Y}$ las compleciones
de $S$, $X$ e $Y$ a lo largo de $S'$, $u^{-1}(S')$ y $v^{-1}(S')$,
respectivamente. Para todo $S$-morfismo $f:X\to Y$, sea
$\widehat{f}:\widehat{X}\to\widehat{Y}$ su extensión a las
compleciones. Entonces la aplicación $f\leadsto\widehat{f}$ es una biyección
\[
  \Hom_S(X,Y)
  \xrightarrow{\ \sim\ }
  \Hom_{\widehat{S}}(\widehat{X},\widehat{Y}).
\]
\end{theorem}

\begin{proof}
Demostremos primero la inyectividad. Supongamos que $f,g:X\to Y$ son
$S$-morfismos tales que $\widehat{f}=\widehat{g}$. Existe entonces
un entorno abierto $V$ de $X'=u^{-1}(S')$ sobre el cual $f$ y $g$
coinciden \sref[I]{I.10.9.4}. Como $u$ es cerrado,
\sref{III.5.1.3.1} da $V=X$, y por tanto $f=g$.

Demostremos ahora la sobreyectividad. Sea
$h:\widehat{X}\to\widehat{Y}$ un
$\widehat{S}$-morfismo. Pongamos
\[
  Z=X\times_S Y,
\]
y denotemos las proyecciones canónicas por $p:Z\to X$ y
$q:Z\to Y$. El preesquema $Z$ es de tipo finito sobre $S$
\sref[I]{I.6.3.4} y, por tanto, noetheriano. Sea
$\widehat{Z}$ su compleción a lo largo de
\[
  Z'=p^{-1}\bigl(u^{-1}(S')\bigr).
\]
El preesquema formal $\widehat{Z}$ se identifica canónicamente con
$\widehat{X}\times_{\widehat{S}}\widehat{Y}$, y sus proyecciones
sobre $\widehat{X}$ y $\widehat{Y}$ se identifican con
$\widehat{p}$ y $\widehat{q}$ \sref[I]{I.10.9.7}.

Como $Y$ es separado sobre $S$, $\widehat{Y}$ es separado sobre
$\widehat{S}$ \sref[I]{I.10.15.7}. El morfismo grafo
\[
  \Gamma_h=(1_{\widehat{X}},h):
  \widehat{X}\longrightarrow\widehat{Z}
\]
es, por consiguiente, una inmersión cerrada \sref[I]{I.10.15.4}. Sea
$\mathfrak{T}$ el subpreesquema formal cerrado de
$\widehat{Z}$ asociado a esta inmersión, y sea
$j:\mathfrak{T}\to\widehat{Z}$ la inmersión canónica. Así,
\[
  \Gamma_h=j\circ w,
\]

$\widehat{p}\circ j$. Además, $\mathfrak{T}$ es propio sobre

\begin{env}[6.6.7]
\label{III.6.6.7}
Vamos a mostrar ahora que el hipertor global definido en (6.6.2), así como
las seis sucesiones espectrales correspondientes, no dependen de los
cubrimientos abiertos afines finitos $\mathfrak U^{(i)}$ que se utilizaron

si $\mathfrak V^{(i)}$ son otros dos cubrimientos de la misma naturaleza,
tales que $\mathfrak V^{(i)}$ sea \emph{más fino} que $\mathfrak U^{(i)}$
para $i=1,2$, entonces se tienen \emph{isomorfismos canónicos} de funtores

\begin{env}[6.6.7, $t$]

\[
  {}^{(t)}E
  (\mathfrak U^{(1)},\mathfrak U^{(2)};
   \mathscr P_\bullet^{(1)},\mathscr P_\bullet^{(2)})
  \xrightarrow{\sim}
  {}^{(t)}E
  (\mathfrak V^{(1)},\mathfrak V^{(2)};
   \mathscr P_\bullet^{(1)},\mathscr P_\bullet^{(2)})
  \tag{6.6.7, $t$)}\label{III.6.6.7.t.eq}
\]
\end{env}
donde $t$ se sustituye por $a$, $b$, $a'$, $b'$, $c$ o $d$.

Ahora bien, para $i=1,2$ se tienen homomorfismos de bicomplejos
\[
  C^\bullet(\mathfrak U^{(i)},\mathscr P_\bullet^{(i)})
  \longrightarrow
  C^\bullet(\mathfrak V^{(i)},\mathscr P_\bullet^{(i)})
\]
bien definidos salvo homotopía (G, II, 5.7.1); de ello resultan ya
homomorfismos (6.6.7, $t$) definidos canónicamente y compatibles con los
operadores de borde en los términos límite ($0_{\mathrm I}$, 11.3.2). Además,
el cálculo de los términos $E_2$ de las sucesiones espectrales $(a)$, $(b)$,
$(a')$, $(b')$ muestra que, para estas sucesiones espectrales, el homomorfismo
(6.6.7, $t$) es un isomorfismo sobre los términos $E_2$; como estas sucesiones
espectrales son birregulares, se ve que (6.6.7, $t$) es un isomorfismo para
estos funtores espectrales y, por consiguiente, induce un \emph{isomorfismo de
bi-$\partial$-funtores} en su término límite común
($0_{\mathrm I}$, 11.1.5).

En particular, para $\mathscr O_{X^{(i)}}$-módulos cuasicoherentes
$\mathscr F^{(i)}$ ($i=1,2$), el homomorfismo canónico
\[
  \mathbf{Tor}_\bullet^S
  (\mathfrak U^{(1)},\mathfrak U^{(2)};
   \mathscr F^{(1)},\mathscr F^{(2)})
  \longrightarrow
  \mathbf{Tor}_\bullet^S
  (\mathfrak V^{(1)},\mathfrak V^{(2)};
   \mathscr F^{(1)},\mathscr F^{(2)})
\]
es biyectivo; visto el cálculo de (6.6.5), se ve que (6.6.7, $t$) es también
un isomorfismo de los términos $E_2$ para $t=c$ y $t=d$. Se concluye como
antes que (6.6.7, $t$) es también un isomorfismo de sucesiones espectrales
para $t=c$ y $t=d$.

\begin{remark}[5.4.3]
\label{III.5.4.3}
Contrariamente a lo que podría sugerir el teorema de existencia
\sref{III.5.1.6}, existen preesquemas formales propios sobre $\Spf(A)$
que \emph{no son algebraizables}, así como existen espacios analíticos
compactos que no proceden de variedades algebraicas complejas. Más
adelante encontraremos tales preesquemas en la «teoría de moduli»,
que, cuando el cuerpo base es $\mathbb{C}$, trata precisamente de las
variaciones infinitesimales de la estructura compleja de una variedad
algebraica completa; es bien sabido que tales variaciones pueden dar
lugar a variedades analíticas que no son algebraicas.
\end{remark}

\begin{proposition}[5.4.4]
\label{III.5.4.4}
Sea $A$ un anillo noetheriano ádico, pongamos
$\mathfrak{S}=\Spf(A)$, y sean
\[
  g:\mathfrak{X}\to\mathfrak{S},
  \qquad
  h:\mathfrak{Y}\to\mathfrak{S}
\]
dos morfismos propios de preesquemas formales. Sea
$f:\mathfrak{X}\to\mathfrak{Y}$ un
$\mathfrak{S}$-morfismo. Si $f$ es finito y
$\mathfrak{Y}$ es algebraizable, entonces $\mathfrak{X}$ es
algebraizable.
\end{proposition}

\begin{proof}
Las hipótesis sobre $g$ y $h$ implican ya que $f$ es propio
\sref{III.3.4.1}; para que $f$ sea finito, basta que la fibra
$f^{-1}(y)$ sea finita para todo $y\in\mathfrak{Y}$
\sref[0]{0.4.8.11}. Por hipótesis,
\[
  \sh{B}=f_*(\sh{O}_{\mathfrak{X}})
\]
es una $\sh{O}_{\mathfrak{Y}}$-álgebra coherente
\sref[0]{0.4.8.6}. Escribamos
$\mathfrak{Y}=\widehat{Y}$ y $h=\widehat{w}$, donde
$w:Y\to\Spec(A)$ es un morfismo propio de preesquemas ordinarios. El
teorema de existencia proporciona una $\sh{O}_Y$-álgebra coherente $\sh{C}$
tal que $\sh{B}=\widehat{\sh{C}}$. Pongamos
\[
  X=\Spec(\sh{C}),
\]
y sea $u:X\to Y$ el morfismo estructural. La definición de
$\mathfrak{X}$ a partir de $\sh{B}$ \sref[0]{0.4.8.7} muestra entonces que
$\mathfrak{X}$ es canónicamente isomorfo a $\widehat{X}$ y
que $f$ se identifica con $\widehat{u}$; basta comprobarlo
cuando $Y$ es afín.
\end{proof}
