\begin{env}[5.1.1]
\label{III.5.1.1}
Sea $A$ un anillo \emph{noetheriano ádico} y sea $\mathfrak{J}$ un ideal de
definición de $A$, de modo que $A$ sea separado y completo para la
topología $\mathfrak{J}$-ádica. Si $Y=\Spec(A)$, el preesquema formal
afín $\Spf(A)$ se identifica con la \emph{compleción}
$\widehat{Y}$ de $Y$ a lo largo del subconjunto cerrado
$Y'=\mathrm{V}(\mathfrak{J})$ \sref[I]{I.10.10.1}. Sea $X$ un
$Y$-preesquema ordinario de tipo finito, de morfismo estructural
$f:X\to Y$. Denotemos por $\widehat{X}$ la \emph{compleción} de $X$ a lo largo
del subconjunto cerrado $X'=f^{-1}(Y')$, o, equivalentemente, el
$\widehat{Y}$-preesquema formal $X\times_Y\widehat{Y}$, y por
$\widehat{f}:\widehat{X}\to\widehat{Y}$ la extensión de $f$ a las
compleciones. Finalmente, para todo $\sh{O}_X$-módulo coherente
$\sh{F}$, denotemos por $\widehat{\sh{F}}$ su compleción
$\sh{F}_{/X'}$, que es un
$\sh{O}_{\widehat{X}}$-módulo coherente.
\end{env}

\begin{proposition}[5.1.2]
\label{III.5.1.2}
Con las hipótesis y la notación de \sref{III.5.1.1}, sea $\sh{F}$
un $\sh{O}_X$-módulo coherente cuyo soporte es propio sobre $Y$
\sref[II]{II.5.4.10}. Los homomorfismos canónicos
\sref{III.4.1.4}
\[
  \rho_i:
  \HH^i(X,\sh{F})
  \longrightarrow
  \HH^i(\widehat{X},\widehat{\sh{F}})
\]
son isomorfismos.
\end{proposition}

\begin{proof}
El $A$-módulo $\HH^i(X,\sh{F})$ es de tipo finito
\sref{III.3.2.4} y, por tanto, se identifica con su compleción
separada para la topología $\mathfrak{J}$-preádica
\sref[0]{0.7.3.6}. La proposición es, por consiguiente, un caso particular
de \sref{III.4.1.10}.
\end{proof}

Recordemos que los isomorfismos canónicos $\rho_i$ conmutan con los
homomorfismos de conexión asociados a toda sucesión exacta
\[
  0\longrightarrow\sh{F}'
  \longrightarrow\sh{F}
  \longrightarrow\sh{F}''
  \longrightarrow0
\]
de $\sh{O}_X$-módulos coherentes
\sref[0]{0.12.1.6} \sref[I]{I.10.8.9}.

\begin{lemma}[5.1.3.1]
\label{III.5.1.3.1}
Bajo las hipótesis de \sref{III.5.1.1}, supongamos además que
$f$ es cerrado. Todo entorno de $X'$ en $X$ es entonces igual

\end{lemma}

\begin{proof}
Nos reducimos primero al caso $f(X)=Y$. En efecto, por hipótesis,
$f(X)$ es un subconjunto cerrado $Z$ de $Y$. Podemos sustituir $f$ por
$f_{\mathrm{red}}$ \sref[I]{I.6.3.4} y suponer así que $X$
e $Y$ son reducidos; podemos entonces sustituir $Y$ por el subpreesquema

anillo cociente de $A$ es, por tanto, ádico y noetheriano.

Se tiene entonces $f(X')=Y'$. Si $V$ es un entorno abierto de $X'$
en $X$, el conjunto $f(X\setmin V)$ es cerrado en $Y$ y no corta a $Y'$.
Esto es imposible salvo que $X\setmin V$ sea vacío, pues $\mathfrak{J}$

Si nos restringimos a los $\sh{O}_X$-módulos coherentes cuyo
soporte es propio sobre $Y$, \sref{III.5.1.3} afirma, en lenguaje
categórico, que el funtor
\[
  \sh{F}\longmapsto\widehat{\sh{F}}
\]
es \emph{plenamente fiel} de la categoría de tales $\sh{O}_X$-módulos en
la categoría de los $\sh{O}_{\widehat{X}}$-módulos coherentes. Proporciona así
una \emph{equivalencia} de la primera categoría con una subcategoría \emph{plena}
de la segunda \sref[0]{0.8.1.6}. El teorema de existencia
mostrará que, cuando $X$ es propio sobre $Y$, esta subcategoría es en
realidad la categoría de \emph{todos} los
$\sh{O}_{\widehat{X}}$-módulos coherentes. Más precisamente:

\begin{theorem}[5.1.4]
\label{III.5.1.4}
Sea $A$ un anillo noetheriano ádico, sea $Y=\Spec(A)$, sea
$\mathfrak{J}$ un ideal de definición de $A$, y pongamos
$Y'=\mathrm{V}(\mathfrak{J})$. Sea $f:X\to Y$ un morfismo separado
de tipo finito y pongamos $X'=f^{-1}(Y')$. Sean
\[
  \widehat{Y}=Y_{/Y'}=\Spf(A),\qquad
  \widehat{X}=X_{/X'}
\]
las compleciones de $Y$ y $X$ a lo largo de $Y'$ y $X'$, y sea
$\widehat{f}:\widehat{X}\to\widehat{Y}$ la extensión de $f$ a
las compleciones. Entonces el funtor
\[
  \sh{F}\longmapsto\sh{F}_{/X'}=\widehat{\sh{F}}
\]
es una equivalencia de la categoría de los
$\sh{O}_X$-módulos coherentes cuyo soporte es propio sobre $\Spec(A)$ en la
categoría de los $\sh{O}_{\widehat{X}}$-módulos coherentes cuyo soporte
es propio sobre $\Spf(A)$.
\end{theorem}

En otras palabras, teniendo en cuenta \sref{III.5.1.3}:

\begin{corollary}[5.1.5]
\label{III.5.1.5}
Un $\sh{O}_{\widehat{X}}$-módulo es isomorfo a la compleción de
un $\sh{O}_X$-módulo coherente cuyo soporte es propio sobre
$\Spec(A)$ si y solo si es coherente y su soporte es propio
sobre $\Spf(A)$.
\end{corollary}

\begin{corollary}[5.1.6]

\end{corollary}

\begin{proposition}[6.5.8]
\label{III.6.5.8}
Sean $X$ e $Y$ dos $S$-preesquemas, y sea $\sh F$
\textup{(respectivamente} $\sh G$\textup{)} un $\sh O_X$-módulo cuasicoherente
\textup{(respectivamente} un $\sh O_Y$-módulo cuasicoherente\textup{)}. Si
$\sh F$ o $\sh G$ es plano sobre $S$, entonces
\[
  \shTor_n^S(\sh F,\sh G)=0\qquad(n\ne0).
\]
\end{proposition}

\begin{proof}
La cuestión es local en $X$ e $Y$, por lo que podemos suponer que
$X$, $Y$ y $S$ son afines, de anillos respectivos $B$, $C$ y
$A$, y escribir $\sh F=\widetilde M$ y $\sh G=\widetilde N$,
donde $M$ es un $B$-módulo y $N$ un $C$-módulo. Supongamos, por
ejemplo, que $\sh F$ es plano sobre $S$. Esto significa que, para
todo $s\in S$, $M_s$ es un $A_s$-módulo plano
(\sref[0]{0.6.7.1}); por consiguiente, $M$ es un $A$-módulo plano
(\sref[0]{0.6.3.3}). Así,
$\Tor_n^A(M,N)=0$ para $n>0$ y todo $C$-módulo $N$
(\sref[0]{0.6.1.1}), y la conclusión se deduce de
\sref{III.6.4.1.1}.
\end{proof}

\begin{lemma}[5.2.2]
\label{III.5.2.2}
Sean $\sh{F}'$ y $\sh{G}'$ $\sh{O}_{\widehat{X}}$-módulos
algebraizables. Para todo homomorfismo
$u:\sh{F}'\to\sh{G}'$, los módulos
$\Ker(u)$, $\Im(u)$ y $\Coker(u)$ son algebraizables.
\end{lemma}

\begin{proof}
Escribamos
\[
  \sh{F}'=\widehat{\sh{F}},\qquad
  \sh{G}'=\widehat{\sh{G}},
\]
donde $\sh{F}$ y $\sh{G}$ son $\sh{O}_X$-módulos coherentes cuyos
soportes son propios sobre $Y$. Por \sref{III.5.1.3},
$u=\widehat{v}$ para un homomorfismo
$v:\sh{F}\to\sh{G}$. Como la compleción es exacta,
$\Ker(\widehat{v})\simeq\widehat{\Ker(v)}$; y como el soporte
de $\Ker(v)$ está contenido en el de $\sh{F}$, se deduce que
$\Ker(u)$ es algebraizable. Los argumentos para $\Im(u)$ y
$\Coker(u)$ son los mismos.
\end{proof}

\begin{proposition}[6.5.13]
\label{III.6.5.13}
Sean $X$ e $Y$ dos $S$-preesquemas, y supongamos que se cumple
una de las condiciones siguientes:
\begin{enumerate}
  \item[\textup{(i)}] $X$ y $Z=X\times_SY$ son localmente noetherianos, y
    $X$ es plano sobre $S$;
  \item[\textup{(ii)}] $S$ y $X$ son localmente noetherianos, y $Y$ es de
    tipo finito sobre $S$.
\end{enumerate}
Sea $\sh P_\bullet$ \textup{(respectivamente} $\sh Q_\bullet$) un complejo
acotado inferiormente de $\sh O_X$-módulos cuasicoherentes
\textup{(respectivamente} de $\sh O_Y$-módulos cuasicoherentes).
Supongamos además que, para todo $n$,
$\sh H_n(\sh P_\bullet)$ \textup{(respectivamente}
$\sh H_n(\sh Q_\bullet)$) es un $\sh O_X$-módulo
\textup{(respectivamente} un $\sh O_Y$-módulo) de tipo finito. Entonces
\[
  \shTor_n^S(\sh P_\bullet,\sh Q_\bullet)
\]
es un $\sh O_Z$-módulo coherente para todo $n$.
\end{proposition}

\begin{proof}
Como $\sh P_\bullet$ y $\sh Q_\bullet$ están acotados
inferiormente, la sucesión espectral
$''\sh E(\sh P_\bullet,\sh Q_\bullet)$ es birregular por
\sref{III.6.5.4}. Tanto en el caso (i) como en el caso (ii), $Z$
es localmente noetheriano, de modo que \sref[0]{0.11.1.8} reduce
la afirmación a la coherencia de los términos
$''\sh E_{pq}^2$. La hipótesis de tipo finito sobre los módulos de
homología y la fórmula \sref{III.6.5.4.2} reducen la proposición
al caso particular en que $\sh P_\bullet$ y $\sh Q_\bullet$ están
concentrados en grado cero, es decir, al corolario siguiente.
\end{proof}

\begin{corollary}[6.5.14]
\label{III.6.5.14}
Supongamos que se cumple una de las condiciones \emph{(i)} y
\emph{(ii)} de \sref{III.6.5.13}, y sea $\sh F$
\textup{(respectivamente} $\sh G$) un $\sh O_X$-módulo cuasicoherente
\textup{(respectivamente} un $\sh O_Y$-módulo cuasicoherente) de tipo
finito. Entonces $\shTor_n^S(\sh F,\sh G)$ es un
$\sh O_Z$-módulo coherente para todo $n$.
\end{corollary}

\begin{proof}

$S$, $X$ e $Y$ son afines.

$\sh L_\bullet$ de $\sh F$, formada por $\sh O_X$-módulos de
tipo finito (\sref[I]{I.1.3.7}). Como $X$ es plano sobre $S$,
\sref{III.6.3.4} da
\[
  \shTor_n^S(\sh F,\sh G)
  =
  \sh H_n(\sh L_\bullet\otimes_S\sh G).
\]

$\sh O_Z$-módulos cuasicoherentes de tipo finito
(\sref[I]{I.9.1.1}), luego son coherentes. Se sigue que
$\sh H_n(\sh L_\bullet\otimes_S\sh G)$ es coherente
(\sref[0]{0.5.3.4}).

\textnormal{(ii)} Supongamos ahora que se cumple la condición
(ii). Como $A(Y)$ es cociente de una $A(S)$-álgebra de polinomios

\end{proof}

\begin{lemma}[5.3.1]
\label{III.5.3.1}
Bajo las condiciones de \sref{III.5.1.4}, sea
\[
  0\longrightarrow\sh{H}\longrightarrow\sh{F}
  \longrightarrow\sh{G}\longrightarrow0
\]
una sucesión exacta de $\sh{O}_{\widehat{X}}$-módulos coherentes.
Si $\sh{G}$ y $\sh{H}$ son algebraizables, entonces $\sh{F}$ es
algebraizable.
\end{lemma}

\begin{proof}
Supongamos que
\[
  \sh{G}=\widehat{\sh{B}},
  \qquad
  \sh{H}=\widehat{\sh{C}},
\]
donde $\sh{B}$ y $\sh{C}$ son $\sh{O}_X$-módulos coherentes cuyos
soportes son propios sobre $Y$. El módulo $\sh{F}$ define canónicamente
un elemento del $A$-módulo
\[
  \Ext^1_{\sh{O}_{\widehat{X}}}
  (\widehat{X};\widehat{\sh{B}},\widehat{\sh{C}})
\]
\sref[0]{0.12.3.2}. Las hipótesis implican que este $A$-módulo es
canónicamente isomorfo a
\[
  \Ext^1_{\sh{O}_X}(X;\sh{B},\sh{C})
\]
\sref{III.4.5.2}. Existe, por tanto, una sucesión exacta
\[
  0\longrightarrow\sh{C}\longrightarrow\sh{A}
  \longrightarrow\sh{B}\longrightarrow0
\]
de $\sh{O}_X$-módulos coherentes cuya clase canónica se envía en la
clase correspondiente a $\sh{F}$. Por definición, teniendo en cuenta
\sref[I]{I.10.8.8}, (ii), esto significa que $\sh{F}$ es isomorfo a
$\widehat{\sh{A}}$. El lema se deduce, pues
$\Supp(\sh{A})$ está contenido en
$\Supp(\sh{B})\cup\Supp(\sh{C})$ y es, por tanto, propio sobre $Y$.
\end{proof}

$\Coker(u)$ son algebraizables, entonces $\sh{F}$ es algebraizable.
\end{corollary}

\begin{proof}

\begin{theorem}[5.4.1]
\label{III.5.4.1}

\begin{proposition}[5.4.4]
\label{III.5.4.4}
Sea $A$ un anillo noetheriano ádico, pongamos
$\mathfrak{S}=\Spf(A)$, y sean
\[
  g:\mathfrak{X}\to\mathfrak{S},
  \qquad
  h:\mathfrak{Y}\to\mathfrak{S}
\]
dos morfismos propios de preesquemas formales. Sea
$f:\mathfrak{X}\to\mathfrak{Y}$ un
$\mathfrak{S}$-morfismo. Si $f$ es finito y
$\mathfrak{Y}$ es algebraizable, entonces $\mathfrak{X}$ es
algebraizable.
\end{proposition}

\begin{proof}
Las hipótesis sobre $g$ y $h$ implican ya que $f$ es propio
\sref{III.3.4.1}; para que $f$ sea finito, basta que la fibra
$f^{-1}(y)$ sea \emph{finita} para todo $y\in\mathfrak{Y}$
\sref[0]{0.4.8.11}. Por hipótesis,
\[
  \sh{B}=f_*(\sh{O}_{\mathfrak{X}})
\]
es una $\sh{O}_{\mathfrak{Y}}$-álgebra coherente
\sref[0]{0.4.8.6}. Escribamos
$\mathfrak{Y}=\widehat{Y}$ y $h=\widehat{w}$, donde
$w:Y\to\Spec(A)$ es un morfismo propio de preesquemas ordinarios. El
teorema de existencia proporciona una $\sh{O}_Y$-álgebra coherente $\sh{C}$
tal que $\sh{B}=\widehat{\sh{C}}$. Pongamos
\[
  X=\Spec(\sh{C}),
\]
y sea $u:X\to Y$ el morfismo estructural. La definición de
$\mathfrak{X}$ a partir de $\sh{B}$ \sref[0]{0.4.8.7} muestra entonces que
$\mathfrak{X}$ es canónicamente isomorfo a $\widehat{X}$ y
que $f$ se identifica con $\widehat{u}$; basta comprobarlo
cuando $Y$ es afín.
\end{proof}

\subsection{Una descomposición de ciertos preesquemas}
\label{subsection:III.5.5}

\begin{proposition}[5.5.1]
\label{III.5.5.1}
Sea $A$ un anillo noetheriano ádico, sea $\mathfrak{J}$ un ideal
de definición de $A$, y pongamos $Y=\Spec(A)$. Sea $f:X\to Y$ un
morfismo separado de tipo finito, y pongamos
\[
  Y_0=\Spec(A/\mathfrak{J}),\qquad
  X_0=X\times_Y Y_0=f^{-1}(Y_0).
\]
Sea $Z_0$ un subconjunto abierto de $X_0$ que es propio sobre $Y_0$.
Entonces existe un subconjunto abierto y cerrado $Z$ de $X$, propio sobre
$Y$, tal que
\[
  Z\cap X_0=Z_0.
\]
\end{proposition}

\begin{proof}
Por hipótesis, existe un subconjunto abierto $T$ de $X$ tal que
$T\cap X_0=Z_0$. Sea $\widehat{T}$ la compleción a lo largo de $Z_0$
del preesquema inducido por $X$ sobre $T$. El soporte de
$\sh{O}_{\widehat{T}}$ es $Z_0$, que es propio sobre $Y_0$;
por tanto, $\widehat{T}$ es propio sobre
$\widehat{Y}=\Spf(A)$ \sref{III.3.4.1}.

Se deduce de \sref{III.5.1.8} que existe un subpreesquema cerrado
$Z$ de $T$, \emph{propio} sobre $Y$, tal que, si
$i:Z\to T$ es la inmersión canónica, entonces
\[
  \widehat{i}:\widehat{Z}\longrightarrow\widehat{T}
\]
es un isomorfismo; aquí $\widehat{Z}$ es la compleción de $Z$
a lo largo de $Z_0$. Por \sref{III.4.6.8}, existe un entorno abierto
$V$ de $Z_0$ en $T$ tal que
$i^{-1}(V)\to V$ es un isomorfismo. Pero $i^{-1}(V)$ es un
entorno de $Z_0$ en $Z$ y, por tanto, es igual a $Z$ por
\sref{III.5.1.3.1}. Así, $Z$ es abierto en $T$, y por tanto en $X$.
También es cerrado en $X$, pues $Z$ es propio sobre $Y$ y $X$ es
separado sobre $Y$. Esto demuestra la proposición.
\end{proof}

\begin{corollary}[5.5.2]
\label{III.5.5.2}
Si $X_0$ es propio sobre $Y_0$, entonces $X$ es la unión de dos
subconjuntos abiertos disjuntos $Z$ y $Z'$ tales que $Z$ es propio sobre
$Y$ y contiene a $X_0$. Además, todo subconjunto cerrado $P$ de $X$
que es propio sobre $Y$ está contenido en $Z$.
\end{corollary}

\begin{proof}
Apliquemos \sref{III.5.5.1} con $Z_0=X_0$, y pongamos $Z'=X-Z$. Para la
última afirmación, $P\cap Z'$ es cerrado en $P$ y, por tanto, propio
sobre $Y$. Si no fuera vacío, $f(P\cap Z')$ sería un subconjunto cerrado
no vacío de $Y$ y, por tanto, cortaría a $Y_0$
\sref{III.5.1.3.1}. Esto contradice
$Z'\cap X_0=\varnothing$ y demuestra que $P\subset Z$.
\end{proof}

\begin{proposition}[6.7.12]
\label{III.6.7.12}
Con las notaciones de (6.7.11), sea $J$ un subconjunto de $I$ tal que, para
$i\in I\setmin J$, se tenga $X^{(i)}=Y^{(i)}=S$, que $f_i$ sea la identidad y que
$\mathscr P_\bullet^{(i)}$ sea el complejo reducido al término de grado $0$
igual a $\mathscr O_S$. Hay entonces un isomorfismo canónico de
$\partial$-funtores

\label{III.6.7.12.1}
\[
  \mathfrak{Tor}_\bullet^S
  ((f_i)_{i\in I};(\mathscr P_\bullet^{(i)})_{i\in I})
  \xrightarrow{\sim}
  \mathfrak{Tor}_\bullet^S
  ((f_i)_{i\in J};(\mathscr P_\bullet^{(i)})_{i\in J}).

\]
\end{env}

\begin{proof}
Podemos limitarnos a definir este isomorfismo cuando $S$ y los $Y^{(i)}$ son
afines, pues el pegado se efectúa como de costumbre. Para $i\in I\setmin J$, puede
tomarse el cubrimiento $\mathfrak U^{(i)}$ formado por el único conjunto $S$,

igual a $\Gamma(S,\mathscr O_S)=A(S)$; el isomorfismo (6.7.12.1) es entonces
evidente.

\begin{corollary}[6.8.3]
\label{III.6.8.3}
Sea $A$ un anillo, sea $S=\Spec(A)$ y sean $X^{(i)}$,
$1\leq i\leq m$, $S$-esquemas cuasicompactos. Para cada $i$, sea
$\sh P_\bullet^{(i)}$ un complejo acotado inferiormente de
$\sh O_{X^{(i)}}$-módulos cuasicoherentes. Existe un funtor
espectral canónico birregular que converge a
\[
  \mathbf{Tor}_\bullet^S
  \bigl(
    X^{(1)},\ldots,X^{(m)};
    \sh P_\bullet^{(1)},\ldots,\sh P_\bullet^{(m)}
  \bigr)
\]
y cuyo término $E_2$ es
\[
\begin{split}
  {}^{(e)}E_{pq}^2
  =
  \bigoplus_{q_1+q_2=q}
  \Tor_p^A
  \Bigl(
    &
    \mathbf{Tor}_{q_1}^S
    \bigl(
      X^{(1)},\ldots,X^{(r)};
      \sh P_\bullet^{(1)},\ldots,\sh P_\bullet^{(r)}
    \bigr),
    \\
    &
    \mathbf{Tor}_{q_2}^S
    \bigl(
      X^{(r+1)},\ldots,X^{(m)};
      \sh P_\bullet^{(r+1)},\ldots,\sh P_\bullet^{(m)}
    \bigr)
  \Bigr)
\end{split}
\]
\end{corollary}

\begin{proof}
Siguiendo la definición de \sref{III.6.6.2}, elijamos para cada
$i$ un recubrimiento abierto afín finito $\mathfrak U^{(i)}$ de
$X^{(i)}$, formemos el bicomplejo
\[
  L_{\bullet\bullet}^{(i)}
  =
  C^\bullet
  \bigl(
    \mathfrak U^{(i)},\sh P_\bullet^{(i)}
  \bigr),
\]
y elijamos una resolución proyectiva de Cartan--Eilenberg
$M_{\bullet\bullet\bullet}^{(i)}$ de este bicomplejo en el sentido
de \sref[0]{0.11.7.1}. El hipertor considerado es la homología del
producto tensorial
\[
  M_{\bullet\bullet\bullet}
  =
  \bigotimes_{i=1}^{m}
  M_{\bullet\bullet\bullet}^{(i)}.
\]
Consideremos $M_{\bullet\bullet\bullet}$ como un complejo simple
$N_\bullet$, producto tensorial de los dos complejos simples
\[
  N_\bullet'
  =
  \bigotimes_{i=1}^{r}
  M_{\bullet\bullet\bullet}^{(i)},
  \qquad
  N_\bullet''
  =
  \bigotimes_{i=r+1}^{m}
  M_{\bullet\bullet\bullet}^{(i)},
\]
donde $N_\bullet'$ y $N_\bullet''$ están graduados por las sumas
de los grados totales de los $M_{\bullet\bullet\bullet}^{(i)}$.

Los $A$-módulos que forman $N_\bullet'$ y $N_\bullet''$ son
\emph{proyectivos}. Por tanto, \sref{III.6.5.9} da
\[
  H_\bullet(M_{\bullet\bullet\bullet})
  =
  \mathbf{Tor}_\bullet^A(N_\bullet',N_\bullet'').
\]
La sucesión espectral buscada es, pues, precisamente
\sref{III.6.3.2.2} aplicada a $N_\bullet'$ y $N_\bullet''$, con
la interpretación precedente de sus módulos de homología para los
dos productos parciales
\[
  X^{(1)}\times_S\ldots\times_SX^{(r)}
  \quad\hbox{y}\quad
  X^{(r+1)}\times_S\ldots\times_SX^{(m)}.
\]
Finalmente, las afirmaciones de regularidad se deducen de
\sref{III.6.3.2}. En efecto, como los
$\sh P_\bullet^{(i)}$ están acotados inferiormente, también lo
están los $M_{\bullet\bullet\bullet}^{(i)}$ y, por consiguiente,
$N_\bullet'$ y $N_\bullet''$.
\end{proof}

\label{III.6.9.4}
Sean $A$ un anillo, $A'$ una $A$-álgebra,
$S=\operatorname{Spec}(A)$, $S'=\operatorname{Spec}(A')$,
$X^{(i)}$ ($1\leq i\leq m$) unos $S$-esquemas cuasicompactos y, para cada
$i$, sea $X'^{(i)}=X_{(S')}^{(i)}$, que es un $S'$-esquema cuasicompacto.
Para cada $i$, sea $\mathscr P_\bullet^{(i)}$ un complejo de
$\mathscr O_{X^{(i)}}$-módulos cuasicoherentes; sea finalmente
$Q'_\bullet$ un complejo de $A'$-módulos, estando estos complejos acotados
inferiormente. Existen tres funtores espectrales birregulares en los
$\mathscr P_\bullet^{(i)}$ y en $Q'_\bullet$, que tienen por término límite
común
\[
  \mathbf{Tor}_\bullet^S
  (X^{(1)},\ldots,X^{(m)},S';
   \mathscr P_\bullet^{(1)},\ldots,\mathscr P_\bullet^{(m)},
   \widetilde{Q'_\bullet})
\]
y cuyos términos $E_2$ son respectivamente
\[
\begin{split}
  {}^{(e)}E_{pq}^2
  =\bigoplus_{q'+q''=q}
  \operatorname{Tor}_p^A\Bigl(
  &\mathbf{Tor}_{q'}^S
  (X^{(1)},\ldots,X^{(m)};
   \mathscr P_\bullet^{(1)},\ldots,\mathscr P_\bullet^{(m)}),\\
  &H_{q''}(Q'_\bullet)
  \Bigr)
\end{split}
\]
\[
\begin{split}
  {}^{(f)}E_{pq}^2
  =\bigoplus_{q_1+q_2+\ldots+q_{m+1}=q}
  \operatorname{Tor}_p^{A'}\Bigl(
  &\mathbf{Tor}_{q_1}^S
  (X^{(1)},S';\mathscr P_\bullet^{(1)},\mathscr O_{S'}),\ldots,\\
  &\ldots,\mathbf{Tor}_{q_m}^S
  (X^{(m)},S';\mathscr P_\bullet^{(m)},\mathscr O_{S'}),
  H_{q_{m+1}}(Q'_\bullet)
  \Bigr)
\end{split}
\]
\[
\begin{split}
  {}^{(f')}E_{pq}^2
  =\bigoplus_{q_1+\ldots+q_m=q}
  \mathbf{Tor}_p^{S'}\Bigl(
  &X'^{(1)},\ldots,X'^{(m)},S';\\

  (\mathscr P_\bullet^{(1)},\mathscr O_{S'}),\ldots,
  \mathscr{T}\!or_{q_m}^S
  (\mathscr P_\bullet^{(m)},\mathscr O_{S'}),
  \widetilde{Q'_\bullet}
  \Bigr).
\end{split}
\]

\begin{proof}
No volveremos sobre el primero de estos funtores espectrales, que fue tratado
en \sref{III.6.8.3} y solo se incluye aquí como recordatorio. Para definir
los otros, consideremos para cada $i$ un cubrimiento abierto afín finito
$\mathfrak U^{(i)}$ de $X^{(i)}$, y, si
$u_i:X'^{(i)}\to X^{(i)}$ es la proyección canónica, el cubrimiento abierto
afín finito correspondiente
$\mathfrak U'^{(i)}=u_i^{-1}(\mathfrak U^{(i)})$.

En virtud de \sref{III.6.6.6},
$\mathbf{Tor}_\bullet^S(X^{(1)},\ldots,X^{(m)},S';
\mathscr P_\bullet^{(1)},\ldots,\mathscr P_\bullet^{(m)},
\widetilde{Q'_\bullet})$ se obtiene tomando, para $1\leq i\leq m$, una
resolución proyectiva de Cartan-Eilenberg
$M_{\bullet\bullet\bullet}^{(i)}$ de
$L_{\bullet\bullet}^{(i)}=C^\bullet(\mathfrak U^{(i)},
\mathscr P_\bullet^{(i)})$ (en el sentido de \sref[0]{0.11.7.1}),
considerando el tricomplejo
\[
  M_{\bullet\bullet\bullet}
  =M_{\bullet\bullet\bullet}^{(1)}\otimes_A
   M_{\bullet\bullet\bullet}^{(2)}\otimes\ldots\otimes_A
   M_{\bullet\bullet\bullet}^{(m)}\otimes_A Q'_\bullet
\]
(donde $Q'_\bullet$ se considera como un tricomplejo cuyos dos últimos
grados se reducen a $0$), y tomando su homología. Si se pone
$M_{\bullet\bullet\bullet}'^{(i)}=
M_{\bullet\bullet\bullet}^{(i)}\otimes_A A'$, se tiene (recordando que
$Q'_\bullet$ es un complejo de $A'$-módulos)
\[
  M_{\bullet\bullet\bullet}
  =M_{\bullet\bullet\bullet}'^{(1)}\otimes_{A'}
   M_{\bullet\bullet\bullet}'^{(2)}\otimes\ldots\otimes_{A'}
   M_{\bullet\bullet\bullet}'^{(m)}\otimes_{A'} Q'_\bullet.
\]
Ahora bien, consideremos cada uno de los complejos
$M_{\bullet\bullet\bullet}'^{(i)}$ como un complejo simple (para su grado
total), y observemos que este complejo está formado por $A'$-módulos
\emph{proyectivos}; de \sref{III.6.3.7} (extendido a un número arbitrario de
complejos) resulta que $H_\bullet(M_{\bullet\bullet\bullet})$ también es
igual a
\[

  (M_{\bullet\bullet\bullet}'^{(1)},\ldots,
   M_{\bullet\bullet\bullet}'^{(m)},Q'_\bullet);
\]
es, por tanto, por \sref{III.6.5.15}, el término límite de una sucesión

cuando $\mathscr P_\bullet^{(i)}$ está acotado inferiormente) y cuyo término
$E_2$ viene dado por
\[

  =\bigoplus_{q_1+\ldots+q_{m+1}=q}
  \operatorname{Tor}_p^{A'}
  (H_{q_1}(M_{\bullet\bullet\bullet}'^{(1)}),\ldots,
   H_{q_m}(M_{\bullet\bullet\bullet}'^{(m)}),H_{q_{m+1}}(Q'_\bullet)).
\]
